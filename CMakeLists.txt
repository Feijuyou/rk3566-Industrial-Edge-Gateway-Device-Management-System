cmake_minimum_required(VERSION 3.16)
project(rk3566_modbus LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra)
# 防止交叉编译时误用 PC 的头文件和库。
if(CMAKE_CROSSCOMPILING AND NOT CMAKE_SYSROOT AND NOT CMAKE_FIND_ROOT_PATH)
    message(FATAL_ERROR "请通过工具链文件设置 CMAKE_SYSROOT/CMAKE_FIND_ROOT_PATH")
endif()
find_package(Threads REQUIRED)
set(MOSQUITTO_ROOT "" CACHE PATH "目标平台 libmosquitto 安装前缀")
find_path(MOSQUITTO_INCLUDE_DIR mosquitto.h HINTS "${MOSQUITTO_ROOT}/include")
find_library(MOSQUITTO_LIBRARY NAMES mosquitto mosquitto_static
             HINTS "${MOSQUITTO_ROOT}/lib" "${MOSQUITTO_ROOT}/lib64")
if(NOT MOSQUITTO_INCLUDE_DIR OR NOT MOSQUITTO_LIBRARY)
    message(FATAL_ERROR "缺少目标平台 libmosquitto 开发文件。见 README_MQTT.md")
endif()
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nlohmann/json.hpp")
    message(FATAL_ERROR "缺少 third_party/nlohmann/json.hpp。见 README_MQTT.md")
endif()
add_library(sqlite3_embedded STATIC third_party/sqlite/sqlite3.c)
target_include_directories(sqlite3_embedded PUBLIC third_party/sqlite)
target_compile_definitions(sqlite3_embedded PRIVATE SQLITE_THREADSAFE=1)
target_link_libraries(sqlite3_embedded PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
if(UNIX)
    target_link_libraries(sqlite3_embedded PUBLIC m)
endif()
add_library(gateway_core STATIC
    src/Database_manager.cpp src/Mqtt_work.cpp src/Mqtt_worker.cpp src/Storage_worker.cpp)
target_include_directories(gateway_core PUBLIC include "${MOSQUITTO_INCLUDE_DIR}"
                           PRIVATE third_party)
target_link_libraries(gateway_core PUBLIC sqlite3_embedded Threads::Threads
                      "${MOSQUITTO_LIBRARY}")
option(MOSQUITTO_STATIC "链接静态 libmosquitto（主要用于本机测试）" OFF)
if(MOSQUITTO_STATIC)
    target_compile_definitions(gateway_core PUBLIC LIBMOSQUITTO_STATIC)
    if(WIN32)
        target_link_libraries(gateway_core PUBLIC ws2_32 pthread)
    endif()
endif()
option(BUILD_GATEWAY_APP "构建 Linux 串口采集主程序" ${UNIX})
if(BUILD_GATEWAY_APP)
    add_executable(modbus_app src/main.cpp src/Modbus_collector.cpp src/Acquisition_worker.cpp)
    target_link_libraries(modbus_app PRIVATE gateway_core)
    option(STATIC_CXX_RUNTIME "静态链接 libstdc++ 和 libgcc" ON)
    if(STATIC_CXX_RUNTIME AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        target_link_options(modbus_app PRIVATE -static-libstdc++ -static-libgcc)
    endif()
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(database_test tests/database_test.cpp)
    target_link_libraries(database_test PRIVATE gateway_core)
    add_test(NAME database COMMAND database_test)
    add_executable(mqtt_test_driver tests/mqtt_test_driver.cpp)
    target_link_libraries(mqtt_test_driver PRIVATE gateway_core)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME mqtt_integration COMMAND "${Python3_EXECUTABLE}"
                 "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_mqtt_integration.py"
                 $<TARGET_FILE:mqtt_test_driver>)
        set_tests_properties(mqtt_integration PROPERTIES TIMEOUT 100)
    endif()
endif()
